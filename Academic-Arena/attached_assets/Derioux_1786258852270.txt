                banner = document.createElement('div');
                banner.className = 'floating-banner';

                return;
            }

            const savedData = localStorage.getItem('academia_save_' + usernameVal);
            if (!savedData) {
                showBanner('Account not found. Please create an account first.', 'login-screen', true);
                return;
            }

            const parsedData = JSON.parse(savedData);
            if (parsedData.password && parsedData.password !== passwordVal) {
                showBanner('Incorrect password. Please try again.', 'login-screen', true);
                return;

            if (!gameState.perspective) gameState.perspective = 'third';

            if (!gameState.strand) {
                goToScreen('strand-screen');
            } else {
                goToScreen('dashboard-screen');
            }

            function castRayDDA(angle) {
                const dirX = Math.cos(angle);
                const dirY = Math.sin(angle);

                let mapX = Math.floor(playerX);
                let mapY = Math.floor(playerY);

                const deltaDistX = Math.abs(1 / (dirX || 0.00001));
                const deltaDistY = Math.abs(1 / (dirY || 0.00001));

                let stepX, stepY, sideDistX, sideDistY;

                if (dirX < 0) {
                    stepX = -1;
                    sideDistX = (playerX - mapX) * deltaDistX;
                } else {
                    stepX = 1;
                    sideDistX = (mapX + 1.0 - playerX) * deltaDistX;
                }
                if (dirY < 0) {
                    stepY = -1;
                    sideDistY = (playerY - mapY) * deltaDistY;
                } else {
                    stepY = 1;
                    sideDistY = (mapY + 1.0 - playerY) * deltaDistY;
                }

                let side = 0;
                let hitVal = 0;
                let maxSteps = 50;

                while (hitVal === 0 && maxSteps > 0) {
                    if (sideDistX < sideDistY) {
                        sideDistX += deltaDistX;
                        mapX += stepX;
                        side = 0;
                    } else {
                        sideDistY += deltaDistY;
                        mapY += stepY;
                        side = 1;
                    }
                    maxSteps--;
                    if (mapX < 0 || mapX >= mapWidth || mapY < 0 || mapY >= mapHeight) {
                        return { dist: 20, side: 0, tile: 1 };
                    }
                    hitVal = map[mapY][mapX];
                }

                let dist;
                if (side === 0) {
                    dist = (mapX - playerX + (1 - stepX) / 2) / dirX;
                } else {
                    dist = (mapY - playerY + (1 - stepY) / 2) / dirY;
                }

                return { dist: Math.max(dist, 0.01), side, tile: hitVal };
            }

            function render() {
                const w = canvas.width;
                const h = canvas.height;
                const fov = Math.PI / 3;

                const ceilGrad = ctx.createLinearGradient(0, 0, 0, h / 2);
                ceilGrad.addColorStop(0, '#070b14');
                ceilGrad.addColorStop(1, '#0f172a');
                ctx.fillStyle = ceilGrad;
                ctx.fillRect(0, 0, w, h / 2);

                const floorGrad = ctx.createLinearGradient(0, h / 2, 0, h);
                floorGrad.addColorStop(0, '#0f172a');
                floorGrad.addColorStop(1, '#070b14');
                ctx.fillStyle = floorGrad;
                ctx.fillRect(0, h / 2, w, h / 2);

                const numRays = Math.min(w, 360);
                const stripWidth = w / numRays;
                const zBuffer = new Array(numRays);

                for (let i = 0; i < numRays; i++) {
                    const rayAngle = playerAngle - fov / 2 + (i / numRays) * fov;
                    const result = castRayDDA(rayAngle);

                    const correctedDist = result.dist * Math.cos(rayAngle - playerAngle);
                    zBuffer[i] = correctedDist;

                    const wallHeight = h / correctedDist;
                    const wallTop = (h - wallHeight) / 2;

                    const brightness = Math.max(0, 1 - correctedDist / 15);
                    let r, g, b;
                    
                    if (result.tile === 2) {
                        r = Math.floor(99 * brightness);
                        g = Math.floor(102 * brightness);
                        b = Math.floor(241 * brightness);
                    } else if (result.tile === 3) {
                        r = Math.floor(56 * brightness);
                        g = Math.floor(189 * brightness);
                        b = Math.floor(248 * brightness);
                    } else {
                        if (result.side === 0) {
                            r = Math.floor(30 * brightness);
                            g = Math.floor(41 * brightness);
                            b = Math.floor(59 * brightness);
                        } else {
                            r = Math.floor(15 * brightness);
                            g = Math.floor(23 * brightness);
                            b = Math.floor(42 * brightness);
                        }
                    }

                    ctx.fillStyle = `rgb(${r},${g},${b})`;
                    ctx.fillRect(Math.floor(i * stripWidth), wallTop, Math.ceil(stripWidth) + 1, wallHeight);
                }

                challengers.sort((a, b) => {
                    const distA = Math.pow(playerX - a.x, 2) + Math.pow(playerY - a.y, 2);
                    const distB = Math.pow(playerX - b.x, 2) + Math.pow(playerY - b.y, 2);
                    return distB - distA;
                });

                challengers.forEach(term => {
                    const spriteX = term.x - playerX;
                    const spriteY = term.y - playerY;
                    
                    const transformX = Math.cos(playerAngle) * spriteX + Math.sin(playerAngle) * spriteY;
                    const transformY = -Math.sin(playerAngle) * spriteX + Math.cos(playerAngle) * spriteY;

                    if (transformX > 0.1) {
                        const spriteScreenX = Math.floor((w / 2) * (1 + transformY / transformX));
                        const spriteSize = Math.abs(Math.floor(h / transformX));
                        const drawX = spriteScreenX - spriteSize * 0.25;
                        const drawY = (h - spriteSize) / 2;

                        const rayIdx = Math.floor((spriteScreenX / w) * numRays);
                        if (rayIdx >= 0 && rayIdx < numRays && transformX < zBuffer[rayIdx]) {
                            const bWidth = spriteSize * 0.5;
                            const bHeight = spriteSize * 0.8;

                            ctx.fillStyle = term.active ? '#1e293b' : '#334155';
                            ctx.fillRect(drawX + bWidth * 0.2, drawY + bHeight * 0.4, bWidth * 0.6, bHeight * 0.6);

                            ctx.fillStyle = term.active ? '#38bdf8' : '#64748b';
                            ctx.fillRect(drawX + bWidth * 0.3, drawY + bHeight * 0.45, bWidth * 0.4, bHeight * 0.3);
                        }
                    }
                });

                // Render Third-Person Avatar if Perspective is Active
                if (gameState.perspective === 'third') {
                    const avatarScreenX = w / 2;
                    const avatarScreenY = h * 0.72;
                    const avatarScale = Math.min(1.8, Math.max(0.6, 120 / (h * 0.35)));

                    ctx.save();
                    ctx.translate(avatarScreenX, avatarScreenY);
                    ctx.scale(avatarScale, avatarScale);

                    // Torso
                    ctx.fillStyle = gameState.avatar.shirt || '#4f46e5';
                    ctx.fillRect(-22, -40, 44, 55);

                    // Head
                    ctx.fillStyle = gameState.avatar.skin || '#f5d0b1';
                    ctx.fillRect(-18, -72, 36, 34);

                    // Hair
                    if (gameState.avatar.hairstyle !== 'bald') {
                        ctx.fillStyle = gameState.avatar.haircolor || '#451a03';
                        ctx.fillRect(-20, -78, 40, 14);
                    }

                    // Legs
                    ctx.fillStyle = gameState.avatar.pants || '#1e293b';
                    ctx.fillRect(-18, 15, 14, 35);
                    ctx.fillRect(4, 15, 14, 35);

                    ctx.restore();
                }

                // Render Minimap
                miniCtx.clearRect(0, 0, 150, 150);
                const miniScale = 150 / mapWidth;
                for (let my = 0; my < mapHeight; my++) {
                    for (let mx = 0; mx < mapWidth; mx++) {
                        if (map[my][mx] > 0) {
                            miniCtx.fillStyle = map[my][mx] === 2 ? '#6366f1' : '#38bdf8';
                            miniCtx.fillRect(mx * miniScale, my * miniScale, miniScale + 0.5, miniScale + 0.5);
                        }
                    }
                }
                miniCtx.fillStyle = '#ef4444';
                challengers.forEach(c => {
                    if (c.active) {
                        miniCtx.fillRect(c.x * miniScale - 3, c.y * miniScale - 3, 6, 6);
                    }
                });
                miniCtx.fillStyle = '#22c55e';
                miniCtx.beginPath();
                miniCtx.arc(playerX * miniScale, playerY * miniScale, 4, 0, Math.PI * 2);
                miniCtx.fill();
            }

            function update(dt) {
                if (!roamingActive) return;

                let moveSpeed = 3.5 * dt;
                let rotSpeed = 2.2 * dt;

                let fwd = 0;
                let strafe = 0;

                if (keys['w'] || keys['arrowup']) fwd += 1;
                if (keys['s'] || keys['arrowdown']) fwd -= 1;
                if (keys['a'] || keys['arrowleft']) strafe -= 1;
                if (keys['d'] || keys['arrowright']) strafe += 1;

                if (leftJoy.active) {
                    fwd -= leftJoy.dy;
                    strafe += leftJoy.dx;
                }

                if (pointerLocked) {
                    playerAngle += mouseDX * 0.003 * window.gameConfig.settings.cameraSensitivity;
                    mouseDX = 0;
                }
                if (rightJoy.active) {
                    playerAngle += rightJoy.dx * 0.05;
                }

                const cos = Math.cos(playerAngle);
                const sin = Math.sin(playerAngle);

                let nextX = playerX + (fwd * cos - strafe * sin) * moveSpeed;
                let nextY = playerY + (fwd * sin + strafe * cos) * moveSpeed;

                const margin = 0.2;
                if (map[Math.floor(playerY)][Math.floor(nextX + (nextX > playerX ? margin : -margin))] === 0) {
                    playerX = nextX;
                }
                if (map[Math.floor(nextY + (nextY > playerY ? margin : -margin))][Math.floor(playerX)] === 0) {
                    playerY = nextY;
                }

                activeTerminalTarget = null;
                let nearTerminal = false;
                challengers.forEach(term => {
                    if (term.active) {
                        const dist = Math.hypot(playerX - term.x, playerY - term.y);
                        if (dist < 1.4) {
                            nearTerminal = true;
                            activeTerminalTarget = term;
                        }
                    }
                });

                if (nearTerminal) {
                    questNotification.style.display = 'block';
                } else {
                    questNotification.style.display = 'none';
                }
            }

            let lastTime = performance.now();
            function gameLoop(now) {
                const dt = (now - lastTime) / 1000;
                lastTime = now;

                if (document.getElementById('game-screen').classList.contains('active')) {
                    update(dt);
                    render();
                }
                requestAnimationFrame(gameLoop);
            }
            requestAnimationFrame(gameLoop);
        }
    </script>
